### 1.2 当前规则问题列表

#### 🔴 问题 1: events 集合允许未认证读写
**位置**: `firestore.rules` line 29-31
```javascript

allow create: if true; // 临时测试用
```
**风险**: 任何人可读取所有活动、可创建垃圾数据

#### 🔴 问题 2: posts 集合允许未认证读取
**位置**: `firestore.rules` line 40-41
```javascript

```
**风险**: 未认证用户可读取所有帖子内容

#### 🔴 问题 3: notifications 集合允许未认证读取
**位置**: `firestore.rules` line 48-49
```javascript

```
**风险**: 未认证用户可读取所有用户通知，严重隐私泄露

#### 🔴 问题 4: conversations 及 messages 允许未认证读取
**位置**: `firestore.rules` line 56-64
```javascript

allow read: if true;  // messages
```
**风险**: 未认证用户可读取所有私聊和群聊消息，**最严重的安全漏洞**

#### 🟡 问题 5: notifications 缺少 userId 字段检查
**位置**: `firestore.rules` line 51-52
```javascript

```
**问题**: 模型中没有 userId 字段（`lib/models/notification.dart`），规则会失效

#### 🟡 问题 6: users 集合允许任何认证用户读取
**位置**: `firestore.rules` line 6
```javascript

```
**风险**: 任何登录用户可读取所有用户资料（可能不符合隐私需求）

### 1.3 修复后的 firestore.rules（完整，可直接替换）

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ============================================
    // 辅助函数
    // ============================================
    
    // 检查用户是否已认证
    function isAuthenticated() {
      return request.auth != null;
    }
    
    // 检查是否是当前用户
    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }
    
    // 检查是否是会话参与者
    function isConversationParticipant(conversationId) {
      return isAuthenticated() && 
        request.auth.uid in get(/databases/$(database)/documents/conversations/$(conversationId)).data.participantIds;
    }
    
    // ============================================
    // 用户集合 - 仅本人可读写
    // ============================================
    match /users/{userId} {
      // 读取：仅本人可读（如需公开资料，可添加 isPublic 字段判断）
      allow read: if isOwner(userId);
      
      // 创建：仅本人可创建自己的文档
      allow create: if isOwner(userId);
      
      // 更新/删除：仅本人可操作
      allow update, delete: if isOwner(userId);
      
      // 用户偏好设置子集合
      match /preferences/{prefId} {
        allow read, write: if isOwner(userId);
      }
    }
    
    // ============================================
    // 活动/事件集合 - 认证用户可读，创建者/参与者可改
    // ============================================
    match /events/{eventId} {
      // 读取：仅认证用户可读
      allow read: if isAuthenticated();
      
      // 创建：仅认证用户可创建，且 creatorId 必须是自己
      allow create: if isAuthenticated() && 
        request.resource.data.creatorId == request.auth.uid;
      
      // 更新：创建者或参与者可更新
      allow update: if isAuthenticated() && (
        resource.data.creatorId == request.auth.uid ||
        request.auth.uid in resource.data.participantIds
      );
      
      // 删除：仅创建者可删除
      allow delete: if isAuthenticated() && 
        resource.data.creatorId == request.auth.uid;
    }
    
    // ============================================
    // 帖子集合 - 认证用户可读，作者可改
    // ============================================
    match /posts/{postId} {
      // 读取：仅认证用户可读
      // 如需公开帖子：allow read: if isAuthenticated() || resource.data.isPublic == true;
      allow read: if isAuthenticated();
      
      // 创建：仅认证用户可创建，且 userId 必须是自己
      allow create: if isAuthenticated() && 
        request.resource.data.userId == request.auth.uid;
      
      // 更新/删除：仅作者可操作
      allow update, delete: if isAuthenticated() && 
        resource.data.userId == request.auth.uid;
    }
    
    // ============================================
    // 通知集合 - 仅接收者可读写
    // ============================================
    match /notifications/{notificationId} {
      // 读取：仅接收者可读
      allow read: if isAuthenticated() && 
        resource.data.receiverId == request.auth.uid;
      
      // 创建：认证用户可创建（通常由后端/Cloud Functions 创建）
      // 如果客户端创建，需校验 receiverId
      allow create: if isAuthenticated();
      
      // 更新：仅接收者可更新（如标记已读）
      allow update: if isAuthenticated() && 
        resource.data.receiverId == request.auth.uid;
      
      // 删除：仅接收者可删除
      allow delete: if isAuthenticated() && 
        resource.data.receiverId == request.auth.uid;
    }
    
    // ============================================
    // 会话集合 - 仅参与者可读写
    // ============================================
    match /conversations/{conversationId} {
      // 读取：仅参与者可读
      allow read: if isAuthenticated() && 
        request.auth.uid in resource.data.participantIds;
      
      // 创建：认证用户可创建，且自己必须在参与者列表中
      allow create: if isAuthenticated() && 
        request.auth.uid in request.resource.data.participantIds;
      
      // 更新：仅参与者可更新
      allow update: if isAuthenticated() && 
        request.auth.uid in resource.data.participantIds;
      
      // 删除：仅参与者可删除（软删除更好）
      allow delete: if isAuthenticated() && 
        request.auth.uid in resource.data.participantIds;
      
      // ----------------------------------------
      // 消息子集合
      // ----------------------------------------
      match /messages/{messageId} {
        // 读取：仅会话参与者可读
        allow read: if isConversationParticipant(conversationId);
        
        // 创建：仅会话参与者可创建，且 senderId 必须是自己
        allow create: if isConversationParticipant(conversationId) && 
          request.resource.data.senderId == request.auth.uid;
        
        // 更新：仅发送者可更新（如编辑消息）
        allow update: if isAuthenticated() && 
          resource.data.senderId == request.auth.uid;
        
        // 删除：仅发送者可删除
        allow delete: if isAuthenticated() && 
          resource.data.senderId == request.auth.uid;
      }
    }
    
    // ============================================
    // activities 集合（如果仍需要，否则可删除）
    // ============================================
    match /activities/{activityId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && 
        request.resource.data.creatorId == request.auth.uid;
      allow update: if isAuthenticated() && (
        resource.data.creatorId == request.auth.uid ||
        request.auth.uid in resource.data.participantIds
      );
      allow delete: if isAuthenticated() && 
        resource.data.creatorId == request.auth.uid;
    }
  }
}
```

```dart
Stream<List<AppNotification>> getNotificationsStream(String userId) {
  return _firestore
      .collection('notifications')
      .where('receiverId', isEqualTo: userId)
      .orderBy('timestamp', descending: true)
      .snapshots()
      .map((snapshot) {
        return snapshot.docs.map((doc) => AppNotification.fromJson(doc.data())).toList();
      });
}
```

### 2.2 统一替代方案

#### 方案：使用 AuthService 单例获取当前用户 ID

**原则**：
1. 禁止 Screen/Service 自行写死用户 ID
2. 统一通过 `AuthService.instance.currentUserId` 获取
3. 如果未登录，返回 null，由调用方处理

```dart
  static final AuthService _instance = AuthService._internal();
```

## 【P0-3 登录与用户文档】

### 3.1 修复前问题分析

#### 当前 Email 登录流程（有缺陷）

```dart
// lib/services/auth_service.dart line 70-83
Future<firebase_auth.UserCredential> signInWithEmailAndPassword(
  String email,
  String password,
) async {
  try {
    final userCredential = await _auth.signInWithEmailAndPassword(
      email: email,
      password: password,
    );
    return userCredential;  // ❌ 未调用 _createOrUpdateUserProfile
  } catch (e) {
    throw Exception('Email sign in failed: $e');
  }
}
```

**问题**：
1. Email 登录成功后，**未调用 `_createOrUpdateUserProfile`**
2. 如果用户在其他设备注册，本设备首次登录时 `users/{uid}` 文档可能不存在
3. 导致后续读取用户资料失败（`getUser(uid)` 返回 null）

#### 会导致的功能异常

1. **用户头像/昵称显示异常**：读取 `users/{uid}` 返回 null
2. **消息发送失败**：创建会话时需要用户资料
3. **Firestore 规则报错**：部分规则依赖用户文档存在

### 3.2 修复方案

#### 实现 `ensureUserDocumentExists(uid)` 方法

```dart
// lib/services/auth_service.dart - 新增方法

/// 确保用户文档存在，不存在则创建
/// 使用 set + merge 避免覆盖已有数据
Future<void> ensureUserDocumentExists(firebase_auth.User user) async {
  final userDoc = await _firebaseService.getUser(user.uid);
  
  if (userDoc == null) {
    // 用户文档不存在，创建新文档
    final name = user.displayName ?? user.email?.split('@').first ?? 'User';
    final username = _generateUsername(name, user.email);
    
    final newUser = app_models.User(
      id: user.uid,
      name: name,
      username: username,
      avatar: user.photoURL ?? 
          'https://ui-avatars.com/api/?name=${Uri.encodeComponent(name)}&background=9333EA&color=fff',
      bio: null,
      location: null,
      isOnline: true,
      lastSeen: DateTime.now(),
      interests: [],
    );
    
    await _firebaseService.addUser(newUser);
  } else {
    // 用户文档已存在，仅更新在线状态
    await _firebaseService.updateUserOnlineStatus(user.uid, true);
  }
}
```

### 3.3 完整登录流程说明

```
用户点击登录按钮
    ↓
LoginScreen._signInWithEmail()
    ↓
AuthService.signInWithEmailAndPassword(email, password)
    ↓
Firebase Auth 验证成功
    ↓
ensureUserDocumentExists(user)
    ├─ 检查 users/{uid} 是否存在
    ├─ 不存在 → 创建新文档（name, username, avatar, isOnline=true）
    └─ 已存在 → 更新 isOnline=true, lastSeen
    ↓
返回 UserCredential
    ↓
AuthWrapper 监听到 authStateChanges
    ↓
```

## 【P0-4 AuthWrapper】

### 4.1 当前问题分析

**位置**: `lib/main.dart` line 144-170


**问题列表**：
1. 每次 build 创建新 `AuthService()` 实例（应使用单例）
2. 无错误处理：`snapshot.hasError` 未处理
3. 无超时处理：网络慢时可能一直 loading
4. 无状态区分：loading/unauthenticated/authenticated/error 混在一起

### 4.2 推荐结构：4 种状态明确处理

```dart
// lib/main.dart - 新的 AuthWrapper

class AuthWrapper extends StatefulWidget {
  const AuthWrapper({super.key});

  @override
  State<AuthWrapper> createState() => _AuthWrapperState();
}

class _AuthWrapperState extends State<AuthWrapper> {
  // 使用单例
  final _authService = AuthService.instance;
  
  // 超时计时器
  Timer? _timeoutTimer;
  bool _isTimeout = false;
  
  @override
  void initState() {
    super.initState();
    // 5 秒超时
    _timeoutTimer = Timer(const Duration(seconds: 5), () {
      if (mounted) {
        setState(() => _isTimeout = true);
      }
    });
  }
  
  @override
  void dispose() {
    _timeoutTimer?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return StreamBuilder<User?>(
      stream: _authService.authStateChanges,
      builder: (context, snapshot) {
        // ========== 状态 1: Loading ==========
        if (snapshot.connectionState == ConnectionState.waiting) {
          // 超时后显示重试选项
          if (_isTimeout) {
            return _buildErrorScreen(
              title: '连接超时',
              message: '无法连接到服务器，请检查网络后重试',
              onRetry: () {
                setState(() {
                  _isTimeout = false;
                  _timeoutTimer?.cancel();
                  _timeoutTimer = Timer(const Duration(seconds: 5), () {
                    if (mounted) setState(() => _isTimeout = true);
                  });
                });
              },
            );
          }
          return _buildLoadingScreen();
        }
        
        // ========== 状态 2: Error ==========
        if (snapshot.hasError) {
          return _buildErrorScreen(
            title: '出错了',
            message: '认证服务异常: ${snapshot.error}',
            onRetry: () {
              // 强制刷新
              setState(() {});
            },
          );
        }
        
        // ========== 状态 3: Authenticated ==========
        if (snapshot.hasData && snapshot.data != null) {
          // 取消超时计时器
          _timeoutTimer?.cancel();
          return const MainScreen();
        }
        
        // ========== 状态 4: Unauthenticated ==========
        _timeoutTimer?.cancel();
        return const LoginScreen();
      },
    );
  }
  
  Widget _buildLoadingScreen() {
    return Scaffold(
      body: Container(
        decoration: const BoxDecoration(
          gradient: LinearGradient(
            begin: Alignment.topLeft,
            end: Alignment.bottomRight,
            colors: [Color(0xFF9333EA), Color(0xFFEC4899)],
          ),
        ),
        child: const Center(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              CircularProgressIndicator(color: Colors.white),
              SizedBox(height: 16),
              Text(
                '正在加载...',
                style: TextStyle(color: Colors.white, fontSize: 16),
              ),
            ],
          ),
        ),
      ),
    );
  }
  
  Widget _buildErrorScreen({
    required String title,
    required String message,
    required VoidCallback onRetry,
  }) {
    return Scaffold(
      body: Center(
        child: Padding(
          padding: const EdgeInsets.all(24),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Icon(
                Icons.error_outline,
                size: 64,
                color: Color(0xFFEF4444),
              ),
              const SizedBox(height: 16),
              Text(
                title,
                style: const TextStyle(
                  fontSize: 20,
                  fontWeight: FontWeight.bold,
                ),
              ),
              const SizedBox(height: 8),
              Text(
                message,
                textAlign: TextAlign.center,
                style: TextStyle(
                  color: Colors.grey[600],
                ),
              ),
              const SizedBox(height: 24),
              ElevatedButton(
                onPressed: onRetry,
                style: ElevatedButton.styleFrom(
                  backgroundColor: const Color(0xFF9333EA),
                  foregroundColor: Colors.white,
                  padding: const EdgeInsets.symmetric(
                    horizontal: 32,
                    vertical: 12,
                  ),
                ),
                child: const Text('重试'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
```

## 【P0-5 Logout】

### 5.1 当前登出流程分析

**位置**: `lib/services/auth_service.dart` line 114-123

```dart
Future<void> signOut() async {
  try {
    await Future.wait([
      _auth.signOut(),
      _googleSignIn.signOut(),
    ]);
  } catch (e) {
    throw Exception('Sign out failed: $e');
  }
}
```

### 5.2 当前风险

1. **在线状态未更新**：
   - 登出后 `users/{uid}.isOnline` 仍为 `true`
   - 其他用户看到该用户"在线"，但实际已登出

2. **Firestore Stream 未取消**：
   - `getEventsStream()`、`getUserConversationsStream()` 等监听未取消
   - 可能导致内存泄漏
   - 登出后仍可能收到数据更新

3. **本地状态未清理**：
   - 如果使用 Provider/State 缓存了用户数据，登出后未清空

### 5.3 修复方案

#### 新的 signOut 实现

```dart
// lib/services/auth_service.dart

/// 安全的登出方法
/// 1. 更新在线状态
/// 2. 清理本地状态
/// 3. 登出 Firebase Auth 和第三方登录
Future<void> signOut() async {
  try {
    final uid = currentUserId;
    
    // Step 1: 更新在线状态（在登出前执行，因为登出后无权限）
    if (uid != null) {
      await _firebaseService.updateUserOnlineStatus(uid, false);
    }
    
    // Step 2: 取消所有 Firestore 监听（如果有全局管理的话）
    // 由于当前没有统一管理，此步骤由各 Screen dispose 处理
    // TODO: 后续可添加 StreamController 统一管理
    
    // Step 3: 登出 Firebase Auth 和第三方登录
    await Future.wait([
      _auth.signOut(),
      _googleSignIn.signOut(),
    ]);
    
    // Step 4: 清理本地缓存（如果有的话）
    // 当前没有本地缓存，未来可添加 SharedPreferences 清理
    
  } catch (e) {
    // 即使更新状态失败，也要尝试登出
    try {
      await _auth.signOut();
      await _googleSignIn.signOut();
    } catch (_) {}
    
    throw Exception('Sign out failed: $e');
  }
}
```

#### 修改 FirebaseService 的 updateUserOnlineStatus

确保登出时正确更新 `lastSeen`：

```dart
// lib/services/firebase_service.dart

Future<void> updateUserOnlineStatus(String userId, bool isOnline) async {
  await _firestore.collection('users').doc(userId).update({
    'isOnline': isOnline,
    'lastSeen': FieldValue.serverTimestamp(),  // ✅ 使用服务器时间戳
  });
}
```

#### 在 Screen 中正确取消监听

各 Screen 应在 `dispose` 中取消 Stream 订阅：

```dart
// 示例：lib/screens/home_screen.dart

class _HomeScreenState extends State<HomeScreen> {
  StreamSubscription<List<Event>>? _eventsSubscription;
  
  @override
  void initState() {
    super.initState();
    // 如果手动订阅 Stream，需要保存引用
  }
  
  @override
  void dispose() {
    // ✅ 取消订阅
    _eventsSubscription?.cancel();
    super.dispose();
  }
}
```

**注意**: 当前代码使用 `StreamBuilder`，它会自动在 Widget dispose 时取消订阅，所以这部分风险较低。但如果未来手动订阅 Stream，需要注意取消。
